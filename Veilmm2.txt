-- ═══ SAFE TELEPORTATION ENGINE ═══
local function safeTeleportToCFrame(targetCFrame)
    local char = LocalPlayer.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum or hum.Health <= 0 then return false end

    if hrp:IsA("BasePart") then
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end

    hrp.CFrame = targetCFrame + Vector3.new(0, 3, 0)
    return true
end

local function findSafeLocation(locationKey)
    if locationKey == "Lobby" then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                local n = obj.Name:lower()
                if n == "lobbyspawn" or n == "lobby" or (obj:IsA("SpawnLocation") and string.match(n, "lobby")) then
                    return obj.CFrame
                end
            end
        end
        return CFrame.new(-108, 140, 82)

    elseif locationKey == "MapHighSpot" then
        local mapFolder = workspace:FindFirstChild("Map") or workspace:FindFirstChildWhichIsA("Model")
        if mapFolder then
            local highestPart = nil
            local maxH = -9999
            for _, obj in ipairs(mapFolder:GetDescendants()) do
                if obj:IsA("BasePart") and obj.CanCollide and not string.match(obj.Name:lower(), "lobby") then
                    if obj.Position.Y > maxH and obj.Position.Y < 300 then
                        maxH = obj.Position.Y
                        highestPart = obj
                    end
                end
            end
            if highestPart then
                return highestPart.CFrame
            end
        end

    elseif locationKey == "DroppedGun" then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") and (obj.Name == "GunDrop" or string.match(obj.Name:lower(), "gundrop")) then
                return obj.CFrame
            end
        end
    end

    return nil
end

local function createTabBtn(name, icon, tabId, order)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 36)
    btn.BackgroundColor3 = (tabId == State.ActiveTab) and C.Surface or C.Sidebar
    btn.BackgroundTransparency = (tabId == State.ActiveTab) and 0 or 1
    btn.BorderSizePixel = 0
    btn.LayoutOrder = order
    btn.Text = ""
    btn.ZIndex = 4
    btn.Parent = TabContainer
    corner(btn, 6)

    local txt = Instance.new("TextLabel")
    txt.Size = UDim2.new(1, -16, 1, 0)
    txt.Position = UDim2.new(0, 12, 0, 0)
    txt.BackgroundTransparency = 1
    txt.Text = (icon or "") .. "  " .. name
    txt.TextColor3 = (tabId == State.ActiveTab) and C.TextPrimary or C.TextSecondary
    txt.TextSize = 12
    txt.Font = Enum.Font.GothamSemibold
    txt.TextXAlignment = Enum.TextXAlignment.Left
    txt.ZIndex = 5
    txt.Parent = btn

    btn.MouseButton1Click:Connect(function()
        State.ActiveTab = tabId
        pageTitle.Text = tabTitleMap[tabId] or name

        for pName, pageObj in pairs(pages) do
            pageObj.Visible = (pName == tabId)
        end

        for _, child in ipairs(TabContainer:GetChildren()) do
            if child:IsA("TextButton") then
                tw(child, {BackgroundTransparency = 1, BackgroundColor3 = C.Sidebar}, 0.2):Play()
                local tLabel = child:FindFirstChildOfClass("TextLabel")
                if tLabel then tw(tLabel, {TextColor3 = C.TextSecondary}, 0.2):Play() end
            end
        end
        tw(btn, {BackgroundTransparency = 0, BackgroundColor3 = C.Surface}, 0.2):Play()
        tw(txt, {TextColor3 = C.TextPrimary}, 0.2):Play()
    end)
    return btn
end

local dragging = false
local dragStart, startPos

Sidebar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

HeaderBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

-- ═══ ROLE & WEAPON DETECTION ═══
local function getPlayerRole(plr)
    if not plr then return "Innocent" end
    local hasKnife = false
    local hasGun = false
    
    local backpack = plr:FindFirstChildOfClass("Backpack")
    if backpack then
        for _, tool in ipairs(backpack:GetChildren()) do
            if tool:IsA("Tool") then
                local tName = tool.Name:lower()
                if tName == "knife" or string.match(tName, "knife") or tool:FindFirstChild("KnifeServer") or tool:FindFirstChild("Knife") or string.match(tName, "blade") then
                    hasKnife = true
                elseif tName == "gun" or string.match(tName, "gun") or string.match(tName, "revolver") or tool:FindFirstChild("GunServer") or string.match(tName, "pistol") then
                    hasGun = true
                end
            end
        end
    end
    
    local char = plr.Character
    if char then
        for _, tool in ipairs(char:GetChildren()) do
            if tool:IsA("Tool") then
                local tName = tool.Name:lower()
                if tName == "knife" or string.match(tName, "knife") or tool:FindFirstChild("KnifeServer") or tool:FindFirstChild("Knife") or string.match(tName, "blade") then
                    hasKnife = true
                elseif tName == "gun" or string.match(tName, "gun") or string.match(tName, "revolver") or tool:FindFirstChild("GunServer") or string.match(tName, "pistol") then
                    hasGun = true
                end
            end
        end
    end
    
    if hasKnife then return "Murderer"
    elseif hasGun then return "Sheriff"
    else return "Innocent" end
end

local function getMurderer()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
            if getPlayerRole(plr) == "Murderer" then
                return plr
            end
        end
    end
    return nil
end

local function isMurdererHoldingKnife(mPlr)
    if not mPlr or not mPlr.Character then return false end
    for _, child in ipairs(mPlr.Character:GetChildren()) do
        if child:IsA("Tool") then
            local tName = child.Name:lower()
            if tName == "knife" or string.match(tName, "knife") or child:FindFirstChild("KnifeServer") or string.match(tName, "blade") then
                return true
            end
        end
    end
    return false
end

local function isHoldingGun()
    local char = LocalPlayer.Character
    if not char then return false end
    for _, child in ipairs(char:GetChildren()) do
        if child:IsA("Tool") then
            local tName = child.Name:lower()
            if tName == "gun" or string.match(tName, "gun") or string.match(tName, "revolver") or child:FindFirstChild("GunServer") then
                return true
            end
        end
    end
    return false
end

-- ═══ MAIN LOOP (HITBOX EXPANDER, ESP, AIM & SAFETY) ═══
RunService.RenderStepped:Connect(function()
    local myChar = LocalPlayer.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
    local murderer = getMurderer()

    -- 1. Murderer Hitbox Expander
    if State.HitboxExpanderEnabled and murderer and murderer.Character then
        local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart")
        if mHRP and mHRP:IsA("BasePart") then
            mHRP.Size = Vector3.new(State.HitboxSize, State.HitboxSize, State.HitboxSize)
            mHRP.Transparency = 0.7
            mHRP.Color = C.MurdererRed
            mHRP.Material = Enum.Material.ForceField
            mHRP.CanCollide = false
        end
    elseif not State.HitboxExpanderEnabled and murderer and murderer.Character then
        local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart")
        if mHRP and mHRP:IsA("BasePart") and mHRP.Size.X > 5 then
            mHRP.Size = Vector3.new(2, 2, 1)
            mHRP.Transparency = 1
        end
    end

    -- 2. Sheriff Aim Assist Camera Lock
    if State.SheriffAimEnabled and State.RMBDown and isHoldingGun() then
        if murderer and murderer.Character then
            local mHRP = murderer.Character:FindFirstChild("Head") or murderer.Character:FindFirstChild("HumanoidRootPart")
            if mHRP then
                local cam = workspace.CurrentCamera
                if cam then
                    cam.CFrame = CFrame.new(cam.CFrame.Position, mHRP.Position)
                end
            end
        end
    end

    -- 3. Murderer Safety (Auto-Evade & Return)
    if State.MurdererSafetyEnabled and myHRP and myHum and myHum.Health > 0 and murderer and murderer.Character then
        local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart")
        if mHRP then
            local dist = (myHRP.Position - mHRP.Position).Magnitude
            local holdingKnife = isMurdererHoldingKnife(murderer)

            if not State.IsEvading then
                if dist < 18 and holdingKnife and (tick() - State.LastEvadeTime > 4) then
                    State.IsEvading = true
                    State.LastEvadeTime = tick()
                    State.SavedMapCFrame = myHRP.CFrame

                    local lobbyCF = findSafeLocation("Lobby")
                    if lobbyCF then
                        safeTeleportToCFrame(lobbyCF)
                    end
                end
            else
                local mDistToSaved = (mHRP.Position - State.SavedMapCFrame.Position).Magnitude
                local isSafeNow = (mDistToSaved > 40) or not holdingKnife or (murderer.Character.Humanoid.Health <= 0)

                if isSafeNow and (tick() - State.LastEvadeTime > 3) then
                    State.IsEvading = false
                    safeTeleportToCFrame(State.SavedMapCFrame)
                end
            end
        end
    end

    -- 4. Role ESP & Overhead Tags System
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local char = plr.Character
            local head = char:FindFirstChild("Head")
            local hrp = char:FindFirstChild("HumanoidRootPart")
            
            if State.RoleESPEnabled and head then
                local role = getPlayerRole(plr)
                local targetColor = C.InnocentGreen
                local roleTitle = "INNOCENT"
                
                if role == "Murderer" then
                    targetColor = C.MurdererRed
                    roleTitle = "🔪 MURDERER"
                elseif role == "Sheriff" then
                    targetColor = C.SheriffBlue
                    roleTitle = "🔫 SHERIFF"
                else
                    roleTitle = "😇 INNOCENT"
                end

                local distStr = ""
                if hrp and myHRP then
                    distStr = " [" .. math.floor((hrp.Position - myHRP.Position).Magnitude) .. "m]"
                end

                -- Overhead Tag
                local bb = char:FindFirstChild("VeilOverheadTag")
                if not bb then
                    bb = Instance.new("BillboardGui")
                    bb.Name = "VeilOverheadTag"
                    bb.Size = UDim2.new(0, 200, 0, 40)
                    bb.StudsOffset = Vector3.new(0, 3, 0)
                    bb.AlwaysOnTop = true
                    bb.Adornee = head
                    bb.Parent = char

                    local tagLabel = Instance.new("TextLabel")
                    tagLabel.Name = "TagText"
                    tagLabel.Size = UDim2.new(1, 0, 1, 0)
                    tagLabel.BackgroundTransparency = 1
                    tagLabel.TextSize = 13
                    tagLabel.Font = Enum.Font.GothamBold
                    tagLabel.TextStrokeTransparency = 0.2
                    tagLabel.TextStrokeColor3 = Color3.new(0, 0, 0)
                    tagLabel.Parent = bb
                end

                local tagText = bb:FindFirstChild("TagText")
                if tagText then
                    tagText.Text = plr.DisplayName .. "\n" .. roleTitle .. distStr
                    tagText.TextColor3 = targetColor
                end

                -- Highlight Body Chams
                local hl = char:FindFirstChild("VeilRoleESP")
                if not hl then
                    hl = Instance.new("Highlight")
                    hl.Name = "VeilRoleESP"
                    hl.FillTransparency = 0.35
                    hl.OutlineTransparency = 0.05
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    hl.Adornee = char
                    hl.Parent = char
                end

                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.FillColor = targetColor
                hl.OutlineColor = targetColor
                hl.Enabled = true

            else
                local hl = char:FindFirstChild("VeilRoleESP")
                if hl then hl:Destroy() end
                
                local bb = char:FindFirstChild("VeilOverheadTag")
                if bb then bb:Destroy() end
            end
        end
    end
end)
